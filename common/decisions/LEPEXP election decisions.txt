### LEP 第二次大选（统一全国后）：独立分类，四党
LEP_second_election_category = {

	LEP_second_election_timer = {
		icon = GFX_decision_generic_political_rally
		priority = 999
		allowed = { always = no }
		available = { hidden_trigger = { always = no } }
		selectable_mission = yes
		is_good = no
		fire_only_once = no
		days_mission_timeout = 90
		timeout_effect = {
			log = "[GetLogRoot]: Decision timeout LEP_second_election_timer"
			if = {
				limit = { check_variable = { LEP_NSU_direct_var = 0 } }
				# 三党竞选：团结联盟未参选
				if = {
					limit = {
						NOT = { check_variable = { LEP_election_DSP_var < LEP_election_KMT_var } }
						NOT = { check_variable = { LEP_election_DSP_var < LEP_election_JNSB_var } }
					}
					set_country_flag = LEP_second_election_DSP_win
				}
				else_if = {
					limit = {
						NOT = { check_variable = { LEP_election_KMT_var < LEP_election_DSP_var } }
						NOT = { check_variable = { LEP_election_KMT_var < LEP_election_JNSB_var } }
					}
					set_country_flag = LEP_second_election_KMT_win
				}
				else = {
					set_country_flag = LEP_second_election_JNSB_win
				}
			}
			else = {
				# 四党竞选：团结联盟按 LEP_NSU_direct_var 的代表参选
				if = {
					limit = {
						NOT = { check_variable = { LEP_election_NSU_var < LEP_election_DSP_var } }
						NOT = { check_variable = { LEP_election_NSU_var < LEP_election_KMT_var } }
						NOT = { check_variable = { LEP_election_NSU_var < LEP_election_JNSB_var } }

				}

					limit = {
						NOT = { check_variable = { LEP_election_DSP_var < LEP_election_KMT_var } }
						NOT = { check_variable = { LEP_election_DSP_var < LEP_election_JNSB_var } }
					}
					set_country_flag = LEP_second_election_DSP_win
				}
				else_if = {
					limit = {
						NOT = { check_variable = { LEP_election_KMT_var < LEP_election_DSP_var } }
						NOT = { check_variable = { LEP_election_KMT_var < LEP_election_JNSB_var } }
					}
					set_country_flag = LEP_second_election_KMT_win
				}
				else = {
					set_country_flag = LEP_second_election_JNSB_win
				}
