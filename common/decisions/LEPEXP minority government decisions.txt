### ─────────────────────────────────────────────────────────────────
### 少数派政府：倒计时展示 + 拉拢党派入阁 + 组建政府
### 挂在「议会法案」分类（LEPEXP_parliament）下。
### 席位不过半（LEP_minority_gov）期间出现。
###   · 拉拢每次花 50 政治点；被查禁/无席位的党也列出，但条件里写明原因。
###   · 凑到 138 席后可点「组建政府」正式组阁、解除少数派状态。
### ★ HOI4 的 decisions 文件必须按分类分组（顶层是分类块，决议嵌在里面）。
### ─────────────────────────────────────────────────────────────────

LEPEXP_parliament = {

	# ── 少数派政府倒计时（KR mission 范式，照 UPC_perils_of_a_minority_government_mission）──
	# days_mission_timeout 由引擎计时；每满 30 天跑一次 timeout_effect 扣罚，然后自我重启。
	# 显示用的天数另存 LEP_minority_days（on_daily 递减），因为 mission 的剩余时间读不到。
	# ── 少数派政府倒计时（照 mod 既有 mission 的写法：LEP_economic_recovery_ongoing）──
	# days + days_mission_timeout：条目挂 360 天、每满 30 天跑一次 timeout_effect 扣罚并自我重启。
	# available 用「仍有少数派状态」判定 —— 点了组建政府后 flag 清掉，条目自动消失。
	LEP_minority_government_timer = {
		icon = GFX_decision_generic_political_discourse
		priority = 999

		# 失败条件写成 NOT 形式，界面上才会读成「没有：少数派政府」（照 LEP_economic_recovery_ongoing）。
		# 席位过半也停扣（否则过半后仍会每 30 天扣一次），但状态要等玩家点「组建政府」才清。
		available = {
		    has_country_flag = LEP_minority_gov_over 
			NOT = {
				check_variable = { LEP_coalition_seats < 138 }
			}
		}
		allowed = { always = no }

		days = 360
		days_mission_timeout = 30
		timeout_effect = {
			add_stability = -0.03
			add_political_power = -40
			# 执政党支持率 -2%
			if = { limit = { has_government = totalist }               add_popularity = { ideology = totalist               popularity = -0.02 } }
			if = { limit = { has_government = syndicalist }            add_popularity = { ideology = syndicalist            popularity = -0.02 } }
			if = { limit = { has_government = radical_socialist }      add_popularity = { ideology = radical_socialist      popularity = -0.02 } }
			if = { limit = { has_government = social_democrat }        add_popularity = { ideology = social_democrat        popularity = -0.02 } }
			if = { limit = { has_government = social_liberal }         add_popularity = { ideology = social_liberal         popularity = -0.02 } }
			if = { limit = { has_government = market_liberal }         add_popularity = { ideology = market_liberal         popularity = -0.02 } }
			if = { limit = { has_government = social_conservative }    add_popularity = { ideology = social_conservative    popularity = -0.02 } }
			if = { limit = { has_government = authoritarian_democrat } add_popularity = { ideology = authoritarian_democrat popularity = -0.02 } }
			if = { limit = { has_government = paternal_autocrat }      add_popularity = { ideology = paternal_autocrat      popularity = -0.02 } }
			if = { limit = { has_government = national_populist }      add_popularity = { ideology = national_populist      popularity = -0.02 } }
			set_variable_to_random = LEP_law_gui_dirty
			activate_mission = LEP_minority_government_timer
		}
	}

	LEP_form_government_decision = {
		icon = GFX_decision_generic_speech
		cost = 0
		fire_only_once = no

		# 与拉拢决议一致：只在少数派政府状态下出现（席位没过半那段时间）
		visible = { has_country_flag = LEP_minority_gov }
		available = {
			has_country_flag = LEP_minority_gov
			check_variable = { LEP_coalition_seats > 137 }
		}
		complete_effect = {
			clr_country_flag = LEP_minority_gov
			set_country_flag = LEP_minority_gov_over   # 供 mission 的失败条件判定
			clear_variable = LEP_minority_days
			set_variable_to_random = LEP_law_gui_dirty
			# 正式组阁：宣布政府成立的一点回报
			add_stability = 0.03
			add_political_power = 20
		}
		ai_will_do = { factor = 0 }
	}

	LEP_invite_syndicalist_decision = {
		# 拉拢用另一个图标，和「组建政府」区分开
		icon = GFX_decision_generic_political_discourse
		cost = 0
		fire_only_once = no

		visible = { has_country_flag = LEP_minority_gov }
		available = {
			NOT = { has_government = syndicalist }
			NOT = { is_in_coalition_with_syndicalist = yes }
			# 该党须在国会占有席位（被查禁的党不参选、席次为 0）——
			# 用 custom_trigger_tooltip 把原因讲清楚，而不是让它整条消失。
			custom_trigger_tooltip = {
				tooltip = LEP_party_has_seats_tt
				check_variable = { LEP_seat_n1 > 0 }
			}
		}
		complete_effect = {
			set_temp_variable = { coalition_partner_var = token:syndicalist }
			add_to_coalition = yes
			LEP_hold_election_seats = yes
		}
		ai_will_do = { factor = 0 }
	}
