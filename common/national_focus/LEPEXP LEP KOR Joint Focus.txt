			if = {
				limit = { NOT = { original_tag = LEP } }
				LEP = {
					exists = yes
					is_subject = no

				}
			}
			if = {
				limit = { NOT = { original_tag = KOR } }
				KOR = {
					exists = yes
					is_subject = yes
				}
			}
		}
		if = {
			limit = { NOT = { original_tag = LEP } }
			LEP = {
				OR = {
					has_subject = JAP
					has_subject = KOR
					has_country_flag = LEPEXP_oversee_JAP
				}
			}
		}
		if = {
			limit = { NOT = { original_tag = KOR } }
			KOR = {
				is_subject_of = LEP
			}
		}
	}

	ai_will_do = { factor = 100 }

	select_effect = {
		LEP = {
			country_event = {
               id = league_jap.20
               hours = 12
            }
		}     
    }

	completion_reward_joint_originator = {
		log = "[GetLogRoot]: completion_reward_joint_originator LEPEXP_LEP_KOR_Joint_JAP_focus"
		effect_tooltip = {
			country_event = {
                id = league_jap.21
                days = 1
            }   
		}
		hidden_effect = { 
			if = {
				limit = {
					original_tag = KOR
				}
				LEP = {
					country_event = {
                        id = league_jap.21
                        days = 1
                    } 
				}
			}
			if = {
				limit = {
					original_tag = LEP
				}
				KOR = {
					country_event = {
                        id = league_jap.21
                        days = 1
                    }

			}
