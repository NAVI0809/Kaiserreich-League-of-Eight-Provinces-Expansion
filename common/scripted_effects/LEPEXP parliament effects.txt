### LEP 国会席位图 · 数据层（274 席）
### 调用 LEP_hold_election_seats：请求重排席位（延迟 1 天执行，见文件末尾说明）
### 实际计算在 LEP_do_count_seats：按各党支持率把席位分下去，产出帧数组 LEP_seat_party
### 影响力效果另叠的席位（其他五党小幅 2 / 大幅 4，青年党 3 / 8）只累加 LEP_seat_pending_N，
###   由 LEP_hold_election_seats 统一兑现进 LEP_seat_bonus_N 后再算进席位；
###   支持率那一笔仍由 party_popularity 自行结算。
### 帧号 = 政治光谱序（10 个意识形态）：
###   0 totalist (#770F00)
###   1 syndicalist (#E30613)
###   2 radical_socialist (#EA5153)
###   3 social_democrat (#C71748)
###   4 social_liberal (#EF7D00)
###   5 market_liberal (#FFED00)
###   6 social_conservative (#005D95)
###   7 authoritarian_democrat (#878787)
###   8 paternal_autocrat (#1D1D1B)
###   9 national_populist (#005321)
### 门槛：支持率不足 0.5% 的党不计席位 —— 界面上显示 0% 的党就不会有席位
### 归一化：有效党派按比例分摊，保证总数恰好 274
### 席位坐标写死在 interface/kaiserreich/LEPEXP_parliament.gui

### ─────────────────────────────────────────────────────────
### LEP_recalc_invite_cost：结盟谈判的政治点价格
###   价格 = 25 + 50 × 已在执政联盟中的党数（不含执政党本身）。
###   决议里写 cost = global.LEP_invite_cost_var，政治点不足时游戏自动禁用该决议。
###   校验点：LEP_do_count_seats 每次重排席位时重算 —— 入盟与换届解散都会触发重排。
### ─────────────────────────────────────────────────────────
LEP_recalc_invite_cost = {
	set_variable = { global.LEP_invite_cost_var = 25 }
	if = { limit = { is_in_coalition_with_totalist = yes } add_to_variable = { global.LEP_invite_cost_var = 50 } }
	if = { limit = { is_in_coalition_with_syndicalist = yes } add_to_variable = { global.LEP_invite_cost_var = 50 } }
	if = { limit = { is_in_coalition_with_radical_socialist = yes } add_to_variable = { global.LEP_invite_cost_var = 50 } }
	if = { limit = { is_in_coalition_with_social_democrat = yes } add_to_variable = { global.LEP_invite_cost_var = 50 } }
	if = { limit = { is_in_coalition_with_social_liberal = yes } add_to_variable = { global.LEP_invite_cost_var = 50 } }
	if = { limit = { is_in_coalition_with_market_liberal = yes } add_to_variable = { global.LEP_invite_cost_var = 50 } }
	if = { limit = { is_in_coalition_with_social_conservative = yes } add_to_variable = { global.LEP_invite_cost_var = 50 } }
	if = { limit = { is_in_coalition_with_authoritarian_democrat = yes } add_to_variable = { global.LEP_invite_cost_var = 50 } }
	if = { limit = { is_in_coalition_with_paternal_autocrat = yes } add_to_variable = { global.LEP_invite_cost_var = 50 } }
	if = { limit = { OR = { is_in_coalition_with_national_populist = yes has_country_flag = LEP_YCP_in_coalition } } add_to_variable = { global.LEP_invite_cost_var = 50 } }
}

LEP_do_count_seats = {
	# 结盟谈判价格随联盟规模递增（详见下方 LEP_recalc_invite_cost）
	LEP_recalc_invite_cost = yes

	# ① 取 10 个意识形态的支持率（KR 里一个意识形态对应一个政党）
	#    注意 party_popularity@xxx 必须包在 { value = ... } 里，否则会被读成 0
	set_temp_variable = { s0 = { value = party_popularity@totalist } }
	set_temp_variable = { s1 = { value = party_popularity@syndicalist } }
	set_temp_variable = { s2 = { value = party_popularity@radical_socialist } }
	set_temp_variable = { s3 = { value = party_popularity@social_democrat } }
	set_temp_variable = { s4 = { value = party_popularity@social_liberal } }
	set_temp_variable = { s5 = { value = party_popularity@market_liberal } }
	set_temp_variable = { s6 = { value = party_popularity@social_conservative } }
	set_temp_variable = { s7 = { value = party_popularity@authoritarian_democrat } }
	set_temp_variable = { s8 = { value = party_popularity@paternal_autocrat } }
	set_temp_variable = { s9 = { value = party_popularity@national_populist } }

	# ①.5 青年党（国家民粹主义）席位接口
	#   LEP_youth_party_seat_var = 百分比修正，缺省 0（+15 即席位份额 +15%，-30 即削减 30%）
	#   用法：add_to_variable = { LEP_youth_party_seat_var = 15 }，改完调用 LEP_hold_election_seats 重排
	#   位置在 ② 之前：未解禁（无 LEP_thaw_broad）时 s9 仍会被归零，变量只在解禁后起作用
	set_temp_variable = { ymod = 1 }
	if = {
		limit = { has_variable = LEP_youth_party_seat_var }
		set_temp_variable = { yadd = LEP_youth_party_seat_var }
		divide_temp_variable = { yadd = 100 }
		add_to_temp_variable = { ymod = yadd }
	}
	# 乘数下限 0：负修正最多把该党支持率压到 0，不会出现负席位
	if = {
		limit = { NOT = { check_variable = { ymod > 0 } } }
		set_temp_variable = { ymod = 0 }
	}
	multiply_temp_variable = { s9 = ymod }

	# ①.6 影响力席位加成（LEP_seat_bonus_0..9）
	#   由 LEP_settle_seat_pending 在 LEP_hold_election_seats 时从 LEP_seat_pending_* 兑现而来，
	#   写入方：其他五党影响力（小幅 2 席 / 大幅 4 席）与青年党（3 / 8）
	#   与本轮 popularity 分到的席位叠加结算：
	#   最终席位 = 按支持率分到的席位 + 本项加成，总数仍为 274
	#   换算：一席 = 有效支持率总和 / 274，把加成折成支持率权重，再走原有归一化
	#   不设单党上下限：写多少就体现多少，仅受 274 席归一化约束
	#   负加成把支持率压到 0 以下时，由 ③ 门槛统一归零，不会出现负席位
	set_temp_variable = { seat_sum = 0 }
	add_to_temp_variable = { seat_sum = s0 }
	add_to_temp_variable = { seat_sum = s1 }
	add_to_temp_variable = { seat_sum = s2 }
	add_to_temp_variable = { seat_sum = s3 }
	add_to_temp_variable = { seat_sum = s4 }
	add_to_temp_variable = { seat_sum = s5 }
	add_to_temp_variable = { seat_sum = s6 }
	add_to_temp_variable = { seat_sum = s7 }
	add_to_temp_variable = { seat_sum = s8 }
	add_to_temp_variable = { seat_sum = s9 }
	if = {
		limit = { check_variable = { seat_sum > 0 } }
		set_temp_variable = { seat_unit = seat_sum }
		divide_temp_variable = { seat_unit = 274 }
		if = {
			limit = {
				has_variable = LEP_seat_bonus_0
				NOT = { check_variable = { LEP_seat_bonus_0 = 0 } }
			}
			set_temp_variable = { sbshift = LEP_seat_bonus_0 }
			multiply_temp_variable = { sbshift = seat_unit }
			add_to_temp_variable = { s0 = sbshift }
		}
		if = {
			limit = {
				has_variable = LEP_seat_bonus_1
				NOT = { check_variable = { LEP_seat_bonus_1 = 0 } }
			}
			set_temp_variable = { sbshift = LEP_seat_bonus_1 }
			multiply_temp_variable = { sbshift = seat_unit }
			add_to_temp_variable = { s1 = sbshift }
		}
		if = {
			limit = {
				has_variable = LEP_seat_bonus_2
				NOT = { check_variable = { LEP_seat_bonus_2 = 0 } }
			}
			set_temp_variable = { sbshift = LEP_seat_bonus_2 }
			multiply_temp_variable = { sbshift = seat_unit }
			add_to_temp_variable = { s2 = sbshift }
		}
		if = {
			limit = {
				has_variable = LEP_seat_bonus_3
				NOT = { check_variable = { LEP_seat_bonus_3 = 0 } }
			}
			set_temp_variable = { sbshift = LEP_seat_bonus_3 }
			multiply_temp_variable = { sbshift = seat_unit }
			add_to_temp_variable = { s3 = sbshift }
		}
		if = {
			limit = {

			}

			add_to_temp_variable = { s4 = sbshift }
		}
		if = {
			limit = {
				has_variable = LEP_seat_bonus_5
				NOT = { check_variable = { LEP_seat_bonus_5 = 0 } }
			}
			set_temp_variable = { sbshift = LEP_seat_bonus_5 }
			multiply_temp_variable = { sbshift = seat_unit }
			add_to_temp_variable = { s5 = sbshift }
		}
		if = {
			limit = {
				has_variable = LEP_seat_bonus_6
				NOT = { check_variable = { LEP_seat_bonus_6 = 0 } }
			}
			set_temp_variable = { sbshift = LEP_seat_bonus_6 }
			multiply_temp_variable = { sbshift = seat_unit }

		}

				has_variable = LEP_seat_bonus_8

			}

			add_to_temp_variable = { s8 = sbshift }
		}

			multiply_temp_variable = { sbshift = seat_unit }
			add_to_temp_variable = { s9 = sbshift }
		}
	}

	# ② 参选资格
	#    常驻：社会民主(3) 社会自由(4) 市场自由(5) 社会保守(6) 威权民主(7)
	#    默认禁、只有「政治解冻及其后果」选宽容才开：
	#    总体主义(0) 工团主义(1) 激进社会主义(2) 家长制独裁(8) 国家民粹主义(9)
	if = {
		limit = { NOT = { has_country_flag = LEP_thaw_broad } }
		set_temp_variable = { s0 = 0 }
		set_temp_variable = { s1 = 0 }
		set_temp_variable = { s8 = 0 }
		set_temp_variable = { s9 = 0 }

			limit = { NOT = { has_country_flag = LEP_lkmt_legalized } }

		}
	}

	# ③ 门槛：不足 0.5% 的党直接不计（界面显示 0% 就真的 0 席）
	if = {
		limit = { check_variable = { s0 < 0.005 } }
		set_temp_variable = { s0 = 0 }
	}
	if = {
		limit = { check_variable = { s1 < 0.005 } }
		set_temp_variable = { s1 = 0 }
	}
	if = {
		limit = { check_variable = { s2 < 0.005 } }
		set_temp_variable = { s2 = 0 }
	}
	if = {
		limit = { check_variable = { s3 < 0.005 } }
		set_temp_variable = { s3 = 0 }
	}
	if = {
		limit = { check_variable = { s4 < 0.005 } }
		set_temp_variable = { s4 = 0 }
	}
	if = {
		limit = { check_variable = { s5 < 0.005 } }
		set_temp_variable = { s5 = 0 }
	}
	if = {
		limit = { check_variable = { s6 < 0.005 } }
		set_temp_variable = { s6 = 0 }
	}
	if = {
		limit = { check_variable = { s7 < 0.005 } }
		set_temp_variable = { s7 = 0 }
	}
	if = {
		limit = { check_variable = { s8 < 0.005 } }
		set_temp_variable = { s8 = 0 }
	}
	if = {
		limit = { check_variable = { s9 < 0.005 } }
		set_temp_variable = { s9 = 0 }
	}

	# ④ 汇总有效支持率
	set_temp_variable = { total = 0 }
	add_to_temp_variable = { total = s0 }
	add_to_temp_variable = { total = s1 }
	add_to_temp_variable = { total = s2 }
	add_to_temp_variable = { total = s3 }
	add_to_temp_variable = { total = s4 }
	add_to_temp_variable = { total = s5 }
	add_to_temp_variable = { total = s6 }
	add_to_temp_variable = { total = s7 }
	add_to_temp_variable = { total = s8 }
	add_to_temp_variable = { total = s9 }

	# ⑤ 归一化并按 274 席换算
	if = {
		limit = { check_variable = { total > 0 } }
		divide_temp_variable = { s0 = total }
	}
	if = {
		limit = { check_variable = { total > 0 } }
		divide_temp_variable = { s1 = total }
	}
	if = {
		limit = { check_variable = { total > 0 } }
		divide_temp_variable = { s2 = total }
	}
	if = {
		limit = { check_variable = { total > 0 } }
		divide_temp_variable = { s3 = total }
	}
	if = {
		limit = { check_variable = { total > 0 } }
		divide_temp_variable = { s4 = total }
	}
	if = {
		limit = { check_variable = { total > 0 } }
		divide_temp_variable = { s5 = total }
	}
	if = {
		limit = { check_variable = { total > 0 } }
		divide_temp_variable = { s6 = total }
	}
	if = {
		limit = { check_variable = { total > 0 } }
		divide_temp_variable = { s7 = total }
	}
	if = {
		limit = { check_variable = { total > 0 } }
		divide_temp_variable = { s8 = total }
	}
	if = {
		limit = { check_variable = { total > 0 } }
		divide_temp_variable = { s9 = total }
	}

	multiply_temp_variable = { s0 = 274 }
	multiply_temp_variable = { s1 = 274 }
	multiply_temp_variable = { s2 = 274 }
	multiply_temp_variable = { s3 = 274 }
	multiply_temp_variable = { s4 = 274 }
	multiply_temp_variable = { s5 = 274 }
	multiply_temp_variable = { s6 = 274 }
	multiply_temp_variable = { s7 = 274 }
	multiply_temp_variable = { s8 = 274 }
	multiply_temp_variable = { s9 = 274 }

	# ⑥ 累加边界（席位 0 号 = 政治光谱最左翼）
	set_temp_variable = { b0 = s0 }
	set_temp_variable = { b1 = b0 }
	add_to_temp_variable = { b1 = s1 }
	set_temp_variable = { b2 = b1 }
	add_to_temp_variable = { b2 = s2 }
	set_temp_variable = { b3 = b2 }
	add_to_temp_variable = { b3 = s3 }
	set_temp_variable = { b4 = b3 }
	add_to_temp_variable = { b4 = s4 }
	set_temp_variable = { b5 = b4 }
	add_to_temp_variable = { b5 = s5 }
	set_temp_variable = { b6 = b5 }
	add_to_temp_variable = { b6 = s6 }
	set_temp_variable = { b7 = b6 }
	add_to_temp_variable = { b7 = s7 }
	set_temp_variable = { b8 = b7 }
	add_to_temp_variable = { b8 = s8 }
	set_temp_variable = { b9 = b8 }
	add_to_temp_variable = { b9 = s9 }

	# ⑦ 逐席填帧
	clear_array = LEP_seat_party
	set_temp_variable = { si = 0 }
	# 同时统计每党实际拿到的席位数（整数），避免浮点四舍五入导致总和 ≠ 274
	set_temp_variable = { c0 = 0 }
	set_temp_variable = { c1 = 0 }
	set_temp_variable = { c2 = 0 }
	set_temp_variable = { c3 = 0 }
	set_temp_variable = { c4 = 0 }
	set_temp_variable = { c5 = 0 }
	set_temp_variable = { c6 = 0 }
	set_temp_variable = { c7 = 0 }
	set_temp_variable = { c8 = 0 }
	set_temp_variable = { c9 = 0 }
	while_loop_effect = {
		limit = { check_variable = { si < 274 } }
		if = {
			limit = { check_variable = { si < b0 } }
			add_to_array = { LEP_seat_party = 0 }
			add_to_temp_variable = { c0 = 1 }
		}
		else_if = {
			limit = { check_variable = { si < b1 } }
			add_to_array = { LEP_seat_party = 1 }
			add_to_temp_variable = { c1 = 1 }
		}
		else_if = {
			limit = { check_variable = { si < b2 } }
			add_to_array = { LEP_seat_party = 2 }
			add_to_temp_variable = { c2 = 1 }
		}
		else_if = {
			limit = { check_variable = { si < b3 } }
			add_to_array = { LEP_seat_party = 3 }
			add_to_temp_variable = { c3 = 1 }
		}
		else_if = {
			limit = { check_variable = { si < b4 } }
			add_to_array = { LEP_seat_party = 4 }
			add_to_temp_variable = { c4 = 1 }
		}
		else_if = {
			limit = { check_variable = { si < b5 } }
			add_to_array = { LEP_seat_party = 5 }
			add_to_temp_variable = { c5 = 1 }
		}
		else_if = {
			limit = { check_variable = { si < b6 } }
			add_to_array = { LEP_seat_party = 6 }
			add_to_temp_variable = { c6 = 1 }
		}
		else_if = {
			limit = { check_variable = { si < b7 } }
			add_to_array = { LEP_seat_party = 7 }
			add_to_temp_variable = { c7 = 1 }
		}
		else_if = {
			limit = { check_variable = { si < b8 } }
			add_to_array = { LEP_seat_party = 8 }
			add_to_temp_variable = { c8 = 1 }
		}
		else_if = {
			limit = { check_variable = { si < b9 } }
			add_to_array = { LEP_seat_party = 9 }
			add_to_temp_variable = { c9 = 1 }
		}
		add_to_temp_variable = { si = 1 }
	}

	set_variable = { LEP_seat_n1 = c1 }
	set_variable = { LEP_seat_n2 = c2 }
	set_variable = { LEP_seat_n3 = c3 }
	set_variable = { LEP_seat_n4 = c4 }
	set_variable = { LEP_seat_n5 = c5 }
	set_variable = { LEP_seat_n6 = c6 }
	set_variable = { LEP_seat_n7 = c7 }
	set_variable = { LEP_seat_n8 = c8 }
	set_variable = { LEP_seat_n9 = c9 }

	# ⑨ 执政联盟总席位：执政党 + 联盟伙伴。同一党只计一次（用 OR 合并两种身份）
	set_variable = { LEP_coalition_seats = 0 }
	if = {
		limit = {
			OR = {
				has_government = totalist
				is_in_coalition_with_totalist = yes
			}

		add_to_variable = { LEP_coalition_seats = LEP_seat_n0 }

	if = {

		limit = {
			OR = {
				has_government = national_populist
				is_in_coalition_with_national_populist = yes
				has_country_flag = LEP_YCP_in_coalition
			}
		}
		add_to_variable = { LEP_coalition_seats = LEP_seat_n9 }
	}

	### ── 席位条：把执政联盟席次折算成 0~28 档（每档约 10 席）供 GUI 取帧 ──
	set_variable = { LEP_seat_bar_frame = LEP_coalition_seats }
	multiply_variable = { LEP_seat_bar_frame = 28 }
	divide_variable = { LEP_seat_bar_frame = 274 }
	round_variable = LEP_seat_bar_frame
	if = { limit = { check_variable = { LEP_seat_bar_frame > 28 } } set_variable = { LEP_seat_bar_frame = 28 } }
	if = { limit = { check_variable = { LEP_seat_bar_frame < 0 } } set_variable = { LEP_seat_bar_frame = 0 } }
	set_variable_to_random = LEP_law_gui_dirty

	### ── 少数派政府检查（席位每次重排后同步一次）────────────────
	### 席位过半 → 清掉少数派状态与倒计时；未过半 → 进入/维持倒计时。
	### 放在这里而不是选举事件里：拉拢党派后席位一变就会立刻重算，
	### 玩家把席位凑过半的那一刻状态自动解除，不必再等下一次选举。
	if = {
		limit = {
			has_country_flag = LEP_dem_gov
			has_country_flag = LEP_seat_enabled
		}
		if = {
			limit = { check_variable = { LEP_coalition_seats < 138 } }
			if = {
				limit = { NOT = { has_country_flag = LEP_minority_gov } }
				set_country_flag = LEP_minority_gov
				clr_country_flag = LEP_minority_gov_over   # 重新进入少数派：清掉上次的「已结束」标记
				set_variable = { LEP_minority_days = 30 }
				activate_mission = LEP_minority_government_timer
			}
		}
		### ── 大选组阁后检查：未过半则提醒组建执政联盟 ──
		### 标记由选举事件落下，结算当天消费掉；已过半则不弹。
		if = {
			limit = { has_country_flag = LEP_election_pending_check }
			clr_country_flag = LEP_election_pending_check
			if = {
				limit = { check_variable = { LEP_coalition_seats < 138 } }
				country_event = { id = lep_election.5 days = 0 }
			}
		}
		### 过半不自动解除 —— 由玩家在国会点击「组建政府」正式组阁后才清。
		### （过半后 on_daily 会暂停扣罚，见 on_actions_LEPEXP.txt）
	}
}

### ─────────────────────────────────────────────────────────
### 请求重排席位。
### 为什么不直接算：调用点常常在同一个效果块里先做 add_popularity /
### set_party_name / add_to_coalition，而同一块内 party_popularity@X 读到的
### 仍是变化前的旧值，当场结算会漏掉刚落下的支持率变化。
### 因此这里只发一个延迟 1 天的事件，等支持率落定后再实际计算。
### 全 mod 的调用点都走这里，故各调用点无需关心顺序问题。
### ─────────────────────────────────────────────────────────

LEP_hold_election_seats = {
	# 席位结算延迟一天（避免同帧读到旧的 party_popularity）。
	# 触发过程藏进 hidden_effect，界面上不露「将触发某事件」；改由下面一行统一提示。
	hidden_effect = {
		if = {
			limit = { has_country_flag = LEP_seat_enabled }
			# 影响力累积先兑现为席位补偿，随后一天重算落席
			LEP_settle_seat_pending = yes
			country_event = { id = LEP_seat_recount.1 days = 1 }
		}
	}
	if = {
		limit = { has_country_flag = LEP_seat_enabled }
		custom_effect_tooltip = LEP_seat_recount_tt
	}
}
