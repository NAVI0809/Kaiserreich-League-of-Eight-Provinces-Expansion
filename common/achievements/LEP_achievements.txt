unique_id = lep_achievements

### 八省联合扩展 · 成就
### 命名风格对齐 KR 原有成就：典故化用、双关、贴着史实的具体意象
### possible 限定 original_tag = LEP；happened 的三段式与 KR 一致

LEP_unfinished_revolution_achievement = {
	possible = {
		custom_override_tooltip = {
			tooltip = LEP_ach_needs_lep
			original_tag = LEP
		}
	}

	happened = {
		if = {
			limit = {
				NOT = { has_global_flag = disallow_achievements_flag }
				OR = {
					NOT = { has_global_flag = on_startup_completed }
					is_ai = no
				}
			}
			custom_override_tooltip = {
				tooltip = LEP_unfinished_revolution_achievement_tooltip
				has_country_flag = LEP_dem_gov
				has_country_flag = LEP_dem_national_stage
			}
		}
		else_if = {
			limit = { has_global_flag = disallow_achievements_flag }
			custom_override_tooltip = {
				tooltip = achievements_disabled
				always = no
			}
		}
		else = {
			custom_override_tooltip = {
				tooltip = achievements_disabled_observer
				always = no
			}
		}
	}
}
