LEP_unfinished_revolution_achievement = {
	possible = {
		custom_override_tooltip = {
			tooltip = LEP_ach_needs_lep
			original_tag = LEP
		}
	}

	happened = {
		if = {
			limit = {
				NOT = { has_global_flag = disallow_achievements_flag }
				OR = {
					NOT = { has_global_flag = on_startup_completed }
					is_ai = no
				}
			}
			custom_override_tooltip = {
				tooltip = LEP_unfinished_revolution_achievement_tooltip
				has_country_flag = LEP_dem_gov
				has_country_flag = LEP_the_unifier   # 统一决议 complete_effect 盖章（见 01 China decisions.txt）
			}
		}
		else_if = {
			limit = { has_global_flag = disallow_achievements_flag }
			custom_override_tooltip = {
				tooltip = achievements_disabled
				always = no
			}
		}
		else = {
			custom_override_tooltip = {
				tooltip = achievements_disabled_observer
				always = no
			}
		}
	}
}

LEP_the_old_state_renewed_achievement = {
	possible = {
		custom_override_tooltip = {
			tooltip = LEP_ach_needs_lep
			original_tag = LEP
		}
	}

	happened = {
		if = {
			limit = {
				NOT = { has_global_flag = disallow_achievements_flag }
				OR = {
					NOT = { has_global_flag = on_startup_completed }
					is_ai = no
				}
			}
			custom_override_tooltip = {
				tooltip = LEP_the_old_state_renewed_achievement_tooltip
				has_completed_focus = LEP_China_Stands_Majestically_in_the_Cosmos
				# 东亚：sinosphere_states（汉字文化圈：中国全境+满洲+日本+朝鲜+台湾+蒙古+新疆+西藏+越老柬）
				# 写法照 KR 的 CHINA_little_trouble_in_big_china_achievement
				all_collection_elements = {
					collection = {
						input = constant:state_groups.sinosphere_states
						name = LEP_the_old_state_renewed_achievement_tooltip
					}
					OR = {   # 自己 / 傀儡 / 盟友 控制皆可
						is_controlled_by_ROOT_or_subject = yes
						is_controlled_by_ROOT_or_ally = yes
					}
				}
				# 其它外岛：沙巴(828)、北婆罗洲(333)、新几内亚西南(843)/西北(669)、南萨哈林(537)、千岛群岛(555)
				NOT = {
					any_state = {
						OR = {
							state = 828
							state = 333
							state = 843
							state = 669
							state = 537
							state = 555
						}
						NOT = {
							OR = {
								is_controlled_by_ROOT_or_subject = yes
								is_controlled_by_ROOT_or_ally = yes
							}
						}
					}
				}
				# 东南亚：不在 sinosphere 组里，逐个查（用现成的东南亚 trigger）
				NOT = {
					any_state = {
						is_southeast_asia = yes
						NOT = {
							OR = {
								is_controlled_by_ROOT_or_subject = yes
								is_controlled_by_ROOT_or_ally = yes
							}
						}
					}
				}
			}
		}
		else_if = {
			limit = { has_global_flag = disallow_achievements_flag }
			custom_override_tooltip = {
				tooltip = achievements_disabled
				always = no
			}
		}
		else = {
			custom_override_tooltip = {
				tooltip = achievements_disabled_observer
				always = no
			}
		}
	}
}
