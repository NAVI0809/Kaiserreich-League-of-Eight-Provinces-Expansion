on_actions = {    
    on_state_control_changed = {
		effect = {
			if = {
				limit = { tag = LEP }
				if = {
					limit = {
						FROM.FROM = { state = 608 } #Beijing
						has_country_flag = LEP_dem_gov
					}
					country_event = { id = league_national.1 days = 3 } 
				}
			}
		}
	}

	on_unit_leader_level_up = {
		effect = {
			if = {
				limit = {
					FROM = { tag = LEP }
				}
				unit_leader_event = league_generic.7 #Champion Event
			}
		}
	}

	on_war_relation_added = {
		effect = {
			if = {
				limit = {
					OR = {
						AND = {
							tag = LEP
							FROM = {
								is_major = yes
								NOT = { tag = CUF }
							}
						}
						AND = {
							NOT = { tag = CUF }
							is_major = yes
							FROM = { tag = LEP }
						}
					}
				}
				LEP = {
					LEP_change_president_clothes_effect = yes
					LEP_war_with_jap_effect = yes
				}
			}
		}
	}

	
}
