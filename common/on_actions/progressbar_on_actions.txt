on_actions = {
	on_daily_LEP = {
		effect = {
			LEP = {
				if = {
					limit = {
						has_country_flag = military_mobilization_add
					}
					add_to_variable = {
						military_mobilization_var = 3.75
					}
					clamp_variable = {
						var = military_mobilization_var
						min = 0
						max = 100
					}
	
				}
				else_if = {
					limit = {
						has_country_flag = military_mobilization_decrease
					}
					add_to_variable = {
						military_mobilization_var = -3.75
					}
					clamp_variable = {
						var = military_mobilization_var
						min = 0
						max = 100
					}	
				}
			}

			LEP = {
				if = {
					limit = {
						has_country_flag = military_mobilization_add_sample
					}
					add_to_variable = {
						military_mobilization_var_sample_2 = 5
					}

					set_variable = { military_mobilization_x = military_mobilization_var_sample_2 }
					divide_variable = { military_mobilization_x = 100 }
					multiply_variable = { military_mobilization_x = 491 }
					add_to_variable = { military_mobilization_x = -8 }
					clamp_variable = { 
						var = military_mobilization_x 
						min = 8 
						max = 464 
					}
	
				}

			
			}
		}
	}
}
